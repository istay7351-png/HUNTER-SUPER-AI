na(z.bx) if showSwept z.bx.set_extend(extend.none) z.bx.set_right(bar_index) z.bx.set_bgcolor(color.new(cGrey, 92)) else z.bx.delete() supply.remove(i) if showSweep label.new(bar_index, high, "⚡",
